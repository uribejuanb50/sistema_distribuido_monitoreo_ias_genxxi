let estructuraReporte = [];
 
// Recorre cada contenedor visual independiente en el tablero de Power BI
document.querySelectorAll('visual-container').forEach((visual, index) => {
    // Intenta obtener el título del gráfico o tabla
    let tituloElem = visual.querySelector('.visualTitle, .header-text, title');
    let titulo = tituloElem ? tituloElem.textContent.trim() : `Visual / Bloque ${index + 1}`;
    // Extrae los textos únicamente dentro de este contenedor
    let textos = Array.from(visual.querySelectorAll('svg text, .pivotTableCellWrap, .tablixCellContent, .value'))
                      .map(nodo => nodo.textContent.trim())
                      .filter(texto => texto !== "");
 
    if (textos.length > 0) {
        estructuraReporte.push({
            'Sección / Visual': titulo,
            'Valores Extraídos': textos
        });
    }
});
